  function build(node, depth) {
    var li = document.createElement("li");
    if (depth >= 3) li.className = "collapsed";   // 默认展开到子分组，叶子（页）收起
    var row = document.createElement("div");
    row.className = "row";

    if (node.children && node.children.length) {
      var tog = document.createElement("span");
      tog.className = "toggle";
      tog.textContent = depth >= 3 ? "▸" : "▾";
      row.appendChild(tog);
      tog.addEventListener("click", function (e) {
        e.stopPropagation();
        var open = li.classList.toggle("collapsed");
        tog.textContent = open ? "▸" : "▾";
      });
    }

    var el = document.createElement(node.h ? "a" : "span");
    el.className = "node cat-" + (node.c || "root") + (node.h ? " leaf" : "");
    el.textContent = node.t;
    if (node.h) el.href = node.h;
    row.appendChild(el);

    li.appendChild(row);

    if (node.children && node.children.length) {
      var ul = document.createElement("ul");
      node.children.forEach(function (ch) { ul.appendChild(build(ch, depth + 1)); });
      li.appendChild(ul);
    }
    return li;
  }

  var root = document.getElementById("mm");
  var ul = document.createElement("ul");
  ul.appendChild(build(TREE, 0));
  root.appendChild(ul);

  function setAll(open) {
    root.querySelectorAll("li").forEach(function (li) {
      if (!li.querySelector("ul")) return;
      li.classList.toggle("collapsed", open);
      var tog = li.querySelector(":scope > .row .toggle");
      if (tog) tog.textContent = open ? "▸" : "▾";
    });
  }
  document.getElementById("mm-expand").addEventListener("click", function () { setAll(false); });
  document.getElementById("mm-collapse").addEventListener("click", function () { setAll(true); });
</script>

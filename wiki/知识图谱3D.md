# 知识图谱 3D（可点击）

下方 3D 力导向图展示全库页面的交叉引用关系。**点击任意节点会跳转到对应页面**。
数据来自 `graph.json`（由 Lint 操作根据页面间的链接自动生成）。

<div id="legend" style="position:absolute;top:12px;left:12px;z-index:10;font:12px/1.6 sans-serif;color:#cfd8e3;background:rgba(15,17,23,.6);padding:8px 10px;border-radius:8px;"></div>
<div id="graph" style="position:relative;height:82vh;border:1px solid #2a2f3a;border-radius:8px;overflow:hidden;"></div>

<script src="https://unpkg.com/3d-force-graph@1.73.2/dist/3d-force-graph.min.js"></script>
<script>
  const COLORS = {
    dotnet: '#6ea8fe', concepts: '#79d17b', standards: '#ffd166',
    patterns: '#c792ea', 'anti-patterns': '#ff6b6b', comparisons: '#4dd0e1',
    governance: '#f78c6c', root: '#f78c6c', sources: '#a5d6a7'
  };

  const Graph = ForceGraph3D()(document.getElementById('graph'))
    .backgroundColor('#0b0e14')
    .nodeLabel('label')
    .nodeAutoColorBy('group')
    .linkColor(() => 'rgba(140,160,200,0.25)')
    .linkOpacity(0.35)
    .linkDirectionalArrowLength(3.2)
    .linkDirectionalArrowRelPos(1)
    .linkDirectionalParticles(1.5)
    .linkDirectionalParticleWidth(1.6)
    .onNodeClick(node => { if (node.link) window.location.href = node.link; });

  Graph.linkDirectionalParticleSpeed(0.006);

  fetch('../graph.json')
    .then(r => r.ok ? r.json() : Promise.reject(r.status))
    .then(data => {
      // 节点大小按连接度数
      const deg = {};
      data.links.forEach(l => { deg[l.source] = (deg[l.source]||0)+1; deg[l.target] = (deg[l.target]||0)+1; });
      data.nodes.forEach(n => { n.val = 1 + (deg[n.id]||0) * 0.9; });
      Graph.nodeColor(n => COLORS[n.group] || '#9aa7b5')
            .nodeVal('val');
      Graph.graphData(data);

      // 图例
      const seen = {};
      const items = data.nodes.map(n => n.group).filter(g => (seen[g] = seen[g] || (seen[g] = true)));
      const labels = { dotnet:'.NET', concepts:'概念', standards:'规范', patterns:'模式',
        'anti-patterns':'反模式', comparisons:'对比', governance:'治理', root:'总览', sources:'资料' };
      document.getElementById('legend').innerHTML = items.map(g =>
        `<div><span style="display:inline-block;width:10px;height:10px;border-radius:50%;background:${COLORS[g]||'#9aa7b5'};margin-right:6px;"></span>${labels[g]||g}</div>`
      ).join('');
    })
    .catch(err => {
      document.getElementById('graph').innerHTML =
        '<p style="color:#ccc;padding:1rem">图谱数据暂未生成（需先运行 Lint 生成 graph.json）。错误：' + err + '</p>';
    });
</script>

> 节点颜色按主题分组（见左上角图例），大小按被引用次数；拖拽旋转、滚轮缩放、点击跳转。图谱随内容增长由 Lint 自动刷新。
